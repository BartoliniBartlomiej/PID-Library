192 [[nodiscard]] T compute(T setpoint, T process_variable) noexcept {
193 // Calculate error: e(t) = setpoint - measurement
194 const T error = setpoint - process_variable;
195
196 // Get time delta
197 const auto current_time = clock_.now();
198 Duration dt{0};
199
200 if (first_run_) {
201 first_run_ = false;
202 last_time_ = current_time;
203 previous_error_ = error;
204 debug_info_.error = error;
205 debug_info_.dt = dt;
206 return T{0};
207 }
208
209 dt = std::chrono::duration_cast<Duration>(current_time - last_time_);
210 last_time_ = current_time;
211
212 // Guard against zero or negative time steps
213 if (dt.count() <= 0) {
214 return previous_output_;
215 }
216
217 const T dt_sec = dt.count();
218
219 // Compute PID based on selected form
220 T output;
221 if (config_.form == PIDForm::Parallel) {
222 output = compute_parallel(error, dt);
223 } else {
224 output = compute_series(error, dt);
225 }
226
227 // Clamp output to limits
228 const T output_clamped = std::clamp(output,
229 config_.output_limits.min_output,
230 config_.output_limits.max_output);
231
232 // Anti-windup: undo last integration step if output is saturated
233 // and error is pushing further into saturation
234 if (config_.enable_anti_windup) {
235 apply_anti_windup(output, output_clamped, error, dt_sec);
236 }
237
238 // Update state for next iteration
239 previous_error_ = error;
240 previous_output_ = output_clamped;
241
242 // Update debug info
243 debug_info_.error = error;
244 debug_info_.output = output_clamped;
245 debug_info_.dt = dt;
246
247 return output_clamped;
248 }

336 [[nodiscard]] T compute_parallel(T error, Duration dt) noexcept {
337 const T dt_sec = dt.count();
338
339 // Proportional term
340 const T p_term = config_.gains.kp * error;
341
342 // Derivative term (with optional filtering)
343 T derivative = (error - previous_error_) / dt_sec;
344 derivative = apply_derivative_filter(derivative, dt);
345 const T d_term = config_.gains.kd * derivative;
346
347 const T p_plus_d = p_term + d_term;
348 const T i_term_prev = config_.gains.ki * integral_;
349 const T output_test = p_plus_d + i_term_prev;
350
351 const bool saturated_high = output_test > config_.output_limits.max_output;
352 const bool saturated_low = output_test < config_.output_limits.min_output;
353 const bool would_worsen = (saturated_high && error > T{0})
354 || (saturated_low && error < T{0});
355
356 if (!config_.enable_anti_windup || !would_worsen) {
357 integral_ += error * dt_sec;
358 }
359
360 const T i_term = config_.gains.ki * integral_;
361
362 // Update debug info
363 debug_info_.p_term = p_term;
364 debug_info_.i_term = i_term;
365 debug_info_.d_term = d_term;
366
367 return p_term + i_term + d_term;
368 }

376 [[nodiscard]] T compute_series(T error, Duration dt) noexcept {
377 const T dt_sec = dt.count();
378
379 // Derivative term
380 T derivative = (error - previous_error_) / dt_sec;
381 derivative = apply_derivative_filter(derivative, dt);
382 T d_contribution = T{0};
383 if (config_.gains.kp > T{0}) {
384 const T Td = config_.gains.kd / config_.gains.kp;
385 d_contribution = Td * derivative;
386 }
387
388 // Integral term — całkowanie warunkowe (anti-windup clamping)
389 T i_contribution_prev = T{0};
390 if (config_.gains.ki > T{0} && config_.gains.kp > T{0}) {
391 const T Ti = config_.gains.kp / config_.gains.ki;
392 i_contribution_prev = integral_ / Ti;
393 }
394
395 const T output_test = config_.gains.kp * (error + i_contribution_prev + d_contribution);
396 const bool saturated_high = output_test > config_.output_limits.max_output;
397 const bool saturated_low = output_test < config_.output_limits.min_output;
398 const bool would_worsen = (saturated_high && error > T{0})
399 || (saturated_low && error < T{0});
400
401 if (!config_.enable_anti_windup || !would_worsen) {
402 integral_ += error * dt_sec;
403 }
404
405 T i_contribution = T{0};
406 if (config_.gains.ki > T{0} && config_.gains.kp > T{0}) {
407 const T Ti = config_.gains.kp / config_.gains.ki;
408 i_contribution = integral_ / Ti;
409 }
410
411 const T output = config_.gains.kp * (error + i_contribution + d_contribution);
412
413 // Update debug info
414 debug_info_.p_term = config_.gains.kp * error;
415 debug_info_.i_term = config_.gains.kp * i_contribution;
416 debug_info_.d_term = config_.gains.kp * d_contribution;
417
418 return output;
419 }

427 [[nodiscard]] T apply_derivative_filter(T raw_derivative, Duration dt) noexcept {
428 if (config_.derivative_filter_coeff <= T{0}) {
429 return raw_derivative;
430 }
431
432 const T alpha = config_.derivative_filter_coeff;
433 const T filtered = alpha * previous_derivative_ + (T{1} - alpha) * raw_derivative;
434 previous_derivative_ = filtered;
435
436 return filtered;
437 }

451 void apply_anti_windup(T raw_output, T clamped_output, T error, T dt_sec) noexcept {
452 const bool saturated_high = raw_output > config_.output_limits.max_output;
453 const bool saturated_low = raw_output < config_.output_limits.min_output;
454
455 if (saturated_high || saturated_low) {
456 // Undo integration only if error is pushing further into saturation
457 if ((saturated_high && error > 0) || (saturated_low && error < 0)) {
458 integral_ -= error * dt_sec;
459 }
460 }
