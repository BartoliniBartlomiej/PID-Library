182 [[nodiscard]] T compute(T setpoint, T process_variable) noexcept {
183 // Calculate error: e(t) = setpoint - measurement
184 const T error = setpoint - process_variable;
185
186 // Get time delta
187 const auto current_time = clock_.now();
188 Duration dt{0};
189
190 if (first_run_) {
191 first_run_ = false;
192 last_time_ = current_time;
193 previous_error_ = error;
194 debug_info_.error = error;
195 debug_info_.dt = dt;
196 return 0; // First run, no valid dt yet
197 }
198
199 dt = std::chrono::duration_cast<Duration>(current_time - last_time_);
200 last_time_ = current_time;
201
202 // Guard against zero or negative time steps
203 if (dt.count() <= 0) {
204 return previous_output_;
205 }
206
207 // Compute PID based on selected form
208 T output;
209 if (config_.form == PIDForm::Parallel) {
210 output = compute_parallel(error, dt);
211 } else {
212 output = compute_series(error, dt);
213 }

217 config_.output_limits.max_output);
218

220 if (config_.enable_anti_windup) {
221 apply_anti_windup(output);
222 }
223
224 // Update state for next iteration

226 previous_output_ = output;

229 debug_info_.error = error;
230 debug_info_.output = output;
231 debug_info_.dt = dt;

322 [[nodiscard]] T compute_parallel(T error, Duration dt) noexcept {
323 const T dt_sec = dt.count();
324
325 // Proportional term
326 const T p_term = config_.gains.kp * error;
327
328 // Integral term (trapezoidal integration)
329 integral_ += error * dt_sec;
330 const T i_term = config_.gains.ki * integral_;
331
332 // Derivative term (with optional filtering)
333 T derivative = (error - previous_error_) / dt_sec;
334 derivative = apply_derivative_filter(derivative, dt);
335 const T d_term = config_.gains.kd * derivative;
336
337 // Update debug info
338 debug_info_.p_term = p_term;
339 debug_info_.i_term = i_term;
340 debug_info_.d_term = d_term;
341
342 return p_term + i_term + d_term;
343 }
344
351 [[nodiscard]] T compute_series(T error, Duration dt) noexcept {
352 const T dt_sec = dt.count();
353
354 // Integral term
355 integral_ += error * dt_sec;
356 T i_contribution = 0;
357 if (config_.gains.ki > 0 && config_.gains.kp > 0) {
358 const T Ti = config_.gains.kp / config_.gains.ki;
359 i_contribution = integral_ / Ti;
360 }
361
362 // Derivative term
363 T derivative = (error - previous_error_) / dt_sec;
364 derivative = apply_derivative_filter(derivative, dt);
365 T d_contribution = 0;
366 if (config_.gains.kp > 0) {
367 const T Td = config_.gains.kd / config_.gains.kp;
368 d_contribution = Td * derivative;
369 }
370
371 const T output = config_.gains.kp * (error + i_contribution + d_contribution);
372
373 // Update debug info
374 debug_info_.p_term = config_.gains.kp * error;
375 debug_info_.i_term = config_.gains.kp * i_contribution;
376 debug_info_.d_term = config_.gains.kp * d_contribution;
377
378 return output;
379 }
380
387 [[nodiscard]] T apply_derivative_filter(T raw_derivative, Duration dt) noexcept {
388 if (config_.derivative_filter_coeff <= 0) {
389 return raw_derivative; // No filtering
390 }
391
392 const T alpha = config_.derivative_filter_coeff;
393 const T filtered = alpha * previous_derivative_ + (1 - alpha) * raw_derivative;
394 previous_derivative_ = filtered;
395
396 return filtered;
397 }
398
404 void apply_anti_windup(T clamped_output) noexcept {
405 // If output was clamped, back-calculate integral
406 const T unclamped_i_term = debug_info_.i_term;
407 const T actual_i_term = clamped_output - debug_info_.p_term - debug_info_.d_term;
408
409 // Update integral to prevent further windup
410 if (config_.gains.ki > 0) {
411 integral_ = actual_i_term / config_.gains.ki;
412 }
413 }
